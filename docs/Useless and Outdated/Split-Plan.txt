ProjectEditor(extends: PaintPower.Editors.Editor): {

    The VM should no longer show in the project editor, it's only for play mode.

    FileEditors: {
        The editors for files will now be called, FileEditors.
        They will extend PaintPower.FileEditors.FileEditor;

        PaintPower.Editors -> PaintPower.FileEditors
        PaintPower.Editors.EditorBase -> PaintPower.FileEditors.FileEditor;

    };

    SpriteEditorHost: {
        The main part to editing sprites.
    };

    SpriteEditors: {

        The editors for sprites (which contains a view for either the Skin editor or the item editor).

    };

    Item editor: {
        The item editor contains the FileExplorer on the left and the FileEditor on the left.
        When a file in the FileExplorer is choosen, we open it in a FileEditor.
    };

    SkinEditor: {
        It's the skin editor.
    };

    SpritePanels: {
        The Sprite Selection panel and the Sprite properties panel.
    };

    PlayerMode(
        When a "Build and Run" button is clicked, the project switches the the player mode,
        clears the stage, and builds the project. There is an option to stop the project, pause and play.
        If there is a build error, it will edit player mode.

        The "Play" button will play the last successfully built version.
        The "Build" button will just build and forget.
        These buttons right now are in the VmPanel, they will be moved to the file menu option "Project".
    );

    HEADER_OPTIONS: ["File", "Edit", "Project", "Help", "Server", "Language"];
};

FolderEditor(extends: PaintPower.Editors.Editor): [
    FileExplorer,
    Tabs,
    FileEditor,
    HEADER_OPTIONS: ["File", "Edit", "Help", "Language"]; // But different options.

WorkspaceEditor(extends: PaintPower.Editors.FolderEditor): {

SingleFileEditor(extends: PaintPower.Editors.FolderEditor): {
    Panel for switching what FileEditor is used.
    ItemEditor,
    HEADER_OPTIONS: ["File", "Edit", "Help", "Language"]; // Different options
}

Here is my version of a lowered program:

// Test.c

#include <stdio.h>

int main(void) {
	int x = 3;
	const int y = 7;
	int z;

	if (x != 3) return 1;

	z = x + y;

	printf("%d", z);

	return 0;
}

// Test.ks
[#script]

[#lang "c"]
[#imports {
	"stdio.h"
}]

[#function "void" "main"](void) [#block]

	[#define var "int" "x"](3);
	[#define const "int" "y"](7);
	[#define var "int" "z"];

	[#if [#get "x"] != 3];
	[#ret](1);

	[#set "z"](x + y);

	([#call addressof("printf")]("%d", [#get "z"]));

[/#block]

@ Test.ksa

@ '|' = instruction end point, the end of an instruction.

bind["c"]
extern "sys:stdio.h"

int _main:
	; declare variables
	c.declareVar   x, 3 	int, mutable|
	c.declareVar   y, 5 	int|
	c.declareVar   z, none, int, mutable|

	cmp 		   mem.getVar x|, notequ, 3|
	if
	block
		ret 1|
	|

	; z = x + y
	; Uses '|' to split instructions.
	mem.setVar 		z, math.add mem.getVar x|, mem.getVar y|||

	std.printf   "%d", z|

	ret 0|
|
